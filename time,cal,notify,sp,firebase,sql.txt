import android.app.TimePickerDialog
import java.util.*


class MainActivity : AppCompatActivity() {

    private lateinit var name:EditText
    private lateinit var age:EditText
    private lateinit var calender:CalendarView
    private lateinit var savebtn:Button
    private lateinit var notify:Button
    private lateinit var time: Button
    private lateinit var txt: TextView
    private var selectedDob:String=""

    val channelID="DOB"

    @SuppressLint("MissingInflatedId")
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContentView(R.layout.activity_main)
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main)) { v, insets ->
            val systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom)
            insets
        }

        name=findViewById(R.id.editTextText)
        age=findViewById(R.id.editTextText1)
        calender=findViewById(R.id.calendarView)
        savebtn=findViewById(R.id.button)
        notify=findViewById(R.id.button1)
        time=findViewById(R.id.button4)
        txt=findViewById(R.id.textView5)

        //time picker
        time.setOnClickListener {
            val cal=Calendar.getInstance()
            val hr=cal.get(Calendar.HOUR_OF_DAY)
            val min=cal.get(Calendar.MINUTE)
            val timePickerDialog= TimePickerDialog(
                this,
                {_,selectedhr,selectedmin->
                val selectedTime= String.format("%02d:%02d",selectedhr,selectedmin)
                txt.text="Time: $selectedTime"
            },hr,min,true)
            timePickerDialog.show()
        }

        //calender
        calender.setOnDateChangeListener { _, year, month, dayOfMonth ->
            selectedDob="$dayOfMonth/${month+1}/$year"
            Toast.makeText(this,"DOB Selected: $selectedDob",Toast.LENGTH_SHORT).show()
        }

        //intent
        savebtn.setOnClickListener{
            SaveDatetoNext()
            val intent=Intent(this,Second::class.java)
            startActivity(intent)
        }

        //notify
        CreateNotify()
        notify.setOnClickListener{
            SendNotify()
        }
    }

    //intent content
    private fun SaveDatetoNext(){
        val name1=name.text.toString()
        val age1=age.text.toString()

        //shared prefference
        val shrdpref=getSharedPreferences("userData", Context.MODE_PRIVATE)
        val editor=shrdpref.edit()
        editor.putString("name",name1)
        editor.putString("age",age1)
        editor.putString("dob",selectedDob)
        editor.apply()

        //firebase
        val databases=FirebaseDatabase.getInstance()
        val ref= databases.getReference("Users")
        val userId=ref.push().key?:return
        val userMap=mapOf(
            "name" to name1,
            "age" to age1,
            "dob" to selectedDob
        )
        ref.child(userId).setValue(userMap)
            .addOnSuccessListener{
            Toast.makeText(this,"saved to firebase", Toast.LENGTH_SHORT).show()
        }
            .addOnFailureListener{
                Toast.makeText(this,"failed to save", Toast.LENGTH_SHORT).show()
            }

        //sqlite
        val db= SqlDb(this)
        val success=db.insertUser(name1,age1,selectedDob)
        if(success){
            Toast.makeText(this,"saved to sql", Toast.LENGTH_SHORT).show()
        }
    }

    //notify
    private fun CreateNotify(){
        if(Build.VERSION.SDK_INT>=Build.VERSION_CODES.TIRAMISU){

            if(ContextCompat.checkSelfPermission(this,Manifest.permission.POST_NOTIFICATIONS)!= PackageManager.PERMISSION_GRANTED){
                requestPermissions(arrayOf(Manifest.permission.POST_NOTIFICATIONS),1)
            }
        }
                val name = "DOB Selection Channel"
                val descriptionText = "Channel for DOB selection notification"
                val importance = NotificationManager.IMPORTANCE_HIGH
                val channel = NotificationChannel(channelID, name, importance).apply {
                    description = descriptionText
                }
                val notificationManager: NotificationManager =
                    getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
                notificationManager.createNotificationChannel(channel)
            }

    private fun SendNotify(){
        var builder=NotificationCompat.Builder(this,channelID)
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setContentTitle("DOB Selected")
            .setContentText("you selected:$selectedDob")
            .setPriority(NotificationCompat.PRIORITY_HIGH)
        val notificationManager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        notificationManager.notify(1, builder.build())
        }

}
